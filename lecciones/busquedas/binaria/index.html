    La búsqueda binaria mira el casillero del medio y descarta la mitad que no puede tenerlo, mientras inicio <= fin.
    
        
    def buscar(lista, x):  # lista ordenada    inicio = 0    fin = len(lista) - 1    while inicio <= fin:  # <=, no <        medio = (inicio + fin) // 2        if lista[medio] == x:            return medio        elif lista[medio] < x:            inicio = medio + 1  # medio ya se miró        else:            fin = medio - 1    return -1  # no está
