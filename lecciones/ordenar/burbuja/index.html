    Burbuja compara vecinos y los intercambia si están al revés: en cada pasada, la mayor sube al final.
    
        
    def burbuja(lista):  # devuelve None    for i in range(len(lista) - 1):        hubo_cambio = False        for j in range(len(lista) - 1 - i):            if lista[j] > lista[j + 1]:  # < : al revés                aux = lista[j]                lista[j] = lista[j + 1]                lista[j + 1] = aux                hubo_cambio = True        if not hubo_cambio:  # ya está ordenada            break# 5 valores: 4 + 3 + 2 + 1 = 10 comparaciones; el doble, unas 4 veces más
