    Para mezclar dos listas ordenadas, mira la primera que queda de cada una, toma la menor y avanza su dedo; al final, pega lo que sobra.
    
        
    def mezclar(a, b):  # a y b, ya ordenadas    resultado = []    i = 0    j = 0    while i < len(a) and j < len(b):  # and, no or        if a[i] <= b[j]:            resultado.append(a[i])            i += 1        else:            resultado.append(b[j])            j += 1    return resultado + a[i:] + b[j:]  # lo que sobra
