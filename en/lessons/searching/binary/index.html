    Binary search checks the middle slot and discards the half that can't have the value, while low <= high.
    
        
    def search(items, x):  # sorted list    low = 0    high = len(items) - 1    while low <= high:  # <=, not <        mid = (low + high) // 2        if items[mid] == x:            return mid        elif items[mid] < x:            low = mid + 1  # mid was already checked        else:            high = mid - 1    return -1  # not there
