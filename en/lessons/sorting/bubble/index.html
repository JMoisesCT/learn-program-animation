    Bubble sort compares neighbors and swaps them if they are out of order: in each pass, the largest one bubbles up to the end.
    
        
    def bubble(items):  # returns None    for i in range(len(items) - 1):        swapped = False        for j in range(len(items) - 1 - i):            if items[j] > items[j + 1]:  # < : high to low                temp = items[j]                items[j] = items[j + 1]                items[j + 1] = temp                swapped = True        if not swapped:  # already sorted            break# 5 values: 4 + 3 + 2 + 1 = 10 comparisons; twice the values, about 4 times as many
