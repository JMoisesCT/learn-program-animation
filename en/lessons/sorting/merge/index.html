    To merge two sorted lists, look at the first one left in each list, take the smaller one and move its pointer; at the end, add the leftovers.
    
        
    def merge(a, b):  # a and b, already sorted    result = []    i = 0    j = 0    while i < len(a) and j < len(b):  # and, not or        if a[i] <= b[j]:            result.append(a[i])            i += 1        else:            result.append(b[j])            j += 1    return result + a[i:] + b[j:]  # the leftovers
